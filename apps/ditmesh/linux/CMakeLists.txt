# Compilation settings that should be applied to most targets.
#
# Be cautious about adding new options here, as plugins use this function by
# default. In most cases, you should add new options to specific targets instead
# of modifying this function.
function(APPLY_STANDARD_SETTINGS TARGET)
  target_compile_features(${TARGET} PUBLIC cxx_std_14)
  target_compile_options(${TARGET} PRIVATE -Wall -Werror)
  target_compile_options(${TARGET} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:-O3>")
  target_compile_definitions(${TARGET} PRIVATE "$<$<NOT:$<CONFIG:Debug>>:NDEBUG>")
endfunction()

  PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/intermediates_do_not_run"
)


# ditmesh only plays a generated sine wave through flutter_soloud and never
# decodes Opus/Ogg/Vorbis/FLAC. Without this the plugin links and bundles its
# prebuilt Xiph libraries -- on Linux a libopus.so.0 that needs glibc 2.43, so
# the plugin fails to load on Ubuntu 24.04 (glibc 2.39). The plugin's
# CMakeLists reads this from the environment of the configure run.
set(ENV{NO_XIPH_LIBS} 1)

# ditmesh: bundle libtim2tox_ffi
# ---------------------------------------------------------------------------
# Install the Tim2Tox FFI shared library into the bundle's lib/ dir so
# `flutter build linux` produces a self-contained bundle. Tim2Tox's Dart loader
# probes <exe dir>/../lib/libtim2tox_ffi.so, which is exactly
# ${INSTALL_BUNDLE_LIB_DIR}. The library comes from
#   tool/ci/build_tim2tox.sh --target linux-<arch>   ->  <repo>/build/native/linux-<arch>/
# (DITMESH_NATIVE_DIR overrides the directory). It links libsodium statically
# by default; with --system-libsodium the captured libsodium.so* travel along
# and the FFI lib's RUNPATH is set to $ORIGIN so they resolve next to it.
# UI tests may explicitly set DITMESH_ALLOW_MISSING_FFI=1. Ordinary builds
# require the native backend.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
  set(_DITMESH_NATIVE_ARCH "aarch64")
else()
  set(_DITMESH_NATIVE_ARCH "x86_64")
endif()
set(DITMESH_NATIVE_DIR
  "${CMAKE_SOURCE_DIR}/../../../build/native/linux-${_DITMESH_NATIVE_ARCH}"
  CACHE PATH "Directory holding libtim2tox_ffi.so (output of tool/ci/build_tim2tox.sh)")
if(DEFINED ENV{DITMESH_NATIVE_DIR})
  set(DITMESH_NATIVE_DIR "$ENV{DITMESH_NATIVE_DIR}")
endif()
set(_ditmesh_ffi_so "${DITMESH_NATIVE_DIR}/libtim2tox_ffi.so")
if(EXISTS "${_ditmesh_ffi_so}")
  install(FILES "${_ditmesh_ffi_so}" DESTINATION "${INSTALL_BUNDLE_LIB_DIR}" COMPONENT Runtime)
  message(STATUS "ditmesh: bundling libtim2tox_ffi.so from ${_ditmesh_ffi_so}")

  # Shared libsodium is only present for --system-libsodium builds; the default
  # pinned build links it statically and this glob is simply empty.
  file(GLOB _ditmesh_sodium_libs "${DITMESH_NATIVE_DIR}/libsodium.so*")
  if(_ditmesh_sodium_libs)
    install(FILES ${_ditmesh_sodium_libs} DESTINATION "${INSTALL_BUNDLE_LIB_DIR}" COMPONENT Runtime)
    message(STATUS "ditmesh: bundling libsodium.so* from ${DITMESH_NATIVE_DIR}")
  endif()

  # runner -> libtim2tox_ffi.so resolves via the runner's $ORIGIN/lib RPATH;
  # libtim2tox_ffi.so -> libsodium.so needs the FFI lib's OWN RUNPATH. patchelf
  # is invoked directly (no shell), so "$ORIGIN" is written verbatim.
  find_program(_DITMESH_PATCHELF patchelf)
  if(_DITMESH_PATCHELF)
    install(CODE
      "execute_process(COMMAND \"${_DITMESH_PATCHELF}\" --set-rpath \"\$ORIGIN\" \"${INSTALL_BUNDLE_LIB_DIR}/libtim2tox_ffi.so\")"
      COMPONENT Runtime)
  elseif(_ditmesh_sodium_libs)
    message(WARNING "ditmesh: patchelf not found; the bundled libsodium next to "
      "libtim2tox_ffi.so may not resolve at runtime (the system libsodium via "
      "ldconfig still works). Install patchelf for a fully self-contained bundle.")
  endif()
elseif("$ENV{DITMESH_ALLOW_MISSING_FFI}" STREQUAL "1")
  message(WARNING "ditmesh: libtim2tox_ffi.so not found at ${_ditmesh_ffi_so}; the "
    "bundle will have NO Tox backend. Run tool/ci/build_tim2tox.sh --target linux "
    "(or ./build_all.sh --platform linux) before `flutter build linux`.")
else()
  message(FATAL_ERROR "DitMesh requires libtim2tox_ffi.so at ${_ditmesh_ffi_so}. "
    "Run tool/ci/build_tim2tox.sh --target linux before building.")
endif()
